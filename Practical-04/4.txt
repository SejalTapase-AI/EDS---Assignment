# 4.2.1 QUESTION:
# 1. Read a CSV filename from user input.
# 2. Columns: Date, Product, Quantity, Price, City.
# 3. Calculate 'Total_Sales' (Quantity * Price).
# 4. Group by Month and find the total sales for each.
# 5. Output the best performing month and its total sales amount.

import pandas as pd

# Prompt the user for the file name
file_name = input()

try:
    # Load the data
    df = pd.read_csv(file_name)

    # Calculate Total Sales for each row
    df['Total_Sales'] = df['Quantity'] * df['Price']

    # Convert Date column to datetime objects
    df['Date'] = pd.to_datetime(df['Date'])

    # Extract the month (Period format YYYY-MM)
    df['Month'] = df['Date'].dt.to_period('M')

    # Group by Month and sum the sales
    monthly_sales = df.groupby('Month')['Total_Sales'].sum()

    # Identify the month with the highest sales
    best_month = monthly_sales.idxmax()
    highest_sales = monthly_sales.max()

    # Display results
    print(f"Best month: {best_month}")
    print(f"Total sales: ${highest_sales:.2f}")

except Exception as e:
    print(f"An error occurred: {e}")

# 4.2.4 QUESTION:
# 1. Read a CSV filename from user input.
# 2. Columns: Date, Product, Quantity, Price, City.
# 3. For each unique date, identify all possible pairs of products sold.
# 4. Count the frequency of each product pair across the entire dataset.
# 5. Output the most frequent product pair(s) and their count.

import pandas as pd
from itertools import combinations
from collections import Counter

# Get filename from user
file_name = input()

try:
    # Load the dataset
    df = pd.read_csv(file_name)

    # Group products by Date
    # We create a list of products for each unique date
    grouped = df.groupby('Date')['Product'].apply(list)

    pair_counts = Counter()

    # Generate pairs for each date
    for products in grouped:
        # Sort products to ensure (A, B) is treated the same as (B, A)
        products.sort()
        # Generate all unique combinations of 2
        unique_pairs = list(combinations(products, 2))
        pair_counts.update(unique_pairs)

    # Find the maximum frequency
    if pair_counts:
        max_freq = max(pair_counts.values())
        
        # Identify all pairs that match the maximum frequency
        best_pairs = [pair for pair, count in pair_counts.items() if count == max_freq]

        # Display results
        for pair in best_pairs:
            # Formatting output to match the requirement: Product A and Product B (Count times)
            print(f"{pair[0]} and {pair[1]} ({max_freq} times)")
    else:
        print("No pairs found.")

except Exception as e:
    print(f"Error: {e}")

# 8. Number of passengers by survival status

# 9. Calculate overall survival rate
survival_rate = data['Survived'].mean()
print(survival_rate)

# 10. Calculate survival rate by gender
survival_by_gender = data.groupby('Sex')['Survived'].mean()
